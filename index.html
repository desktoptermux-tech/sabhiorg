Setup guide — connect to Google Sheets

1

Create a new Google Sheet

Go to sheets.google.com and create a new spreadsheet. Name it "Vehicle Insurance" or anything you like.
2

Open Apps Script

In your sheet, click Extensions → Apps Script. A script editor tab will open.
3

Paste the backend code

Delete all existing code in the editor. Open the Code.gs file provided with this package and paste its entire contents. Click Save (Ctrl+S).
4

Deploy as a web app

Click Deploy → New deployment. Choose type Web app. Set "Execute as" → Me. Set "Who has access" → Anyone. Click Deploy and authorise when prompted.
5

Copy the web app URL

After deployment you'll see a URL like https://script.google.com/macros/s/ABC.../exec. Copy it.
✓

Paste the URL above and click Connect

Paste the URL into the yellow banner at the top and click Connect. Your data will now sync live with Google Sheets.

Note: Each time you change the Apps Script code, create a new deployment (not "Manage existing"). The URL changes with each new deployment — update it in the banner.

// ============================================================ // Vehicle Insurance Manager — Google Apps Script Backend // Paste this entire file into your Apps Script editor // ============================================================ const SHEET_NAME = "VehicleInsurance"; // Column order in the sheet const COLUMNS = [ "id", "reg", "owner", "vtype", "make", "model", "year", "engine", "chassis", "insurer", "policyNo", "ptype", "status", "premium", "idv", "ncb", "agent", "start", "expiry", "notes", "createdAt" ]; // ── Helpers ────────────────────────────────────────────────── function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(COLUMNS); // write header row sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, COLUMNS.length) .setFontWeight("bold") .setBackground("#185FA5") .setFontColor("#ffffff"); sheet.setColumnWidth(1, 160); // id sheet.setColumnWidth(2, 120); // reg sheet.setColumnWidth(3, 150); // owner } return sheet; } function rowToObj(row) { const obj = {}; COLUMNS.forEach((col, i) => obj[col] = row[i] !== undefined ? row[i].toString() : ""); return obj; } function objToRow(obj) { return COLUMNS.map(col => obj[col] !== undefined ? obj[col] : ""); } function getAllRows(sheet) { const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; // only header or empty return data.slice(1).map(rowToObj); // skip header } function findRowIndex(sheet, id) { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0].toString() === id.toString()) return i + 1; // 1-indexed } return -1; } function corsResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ── GET — return all records ────────────────────────────────── function doGet(e) { try { const sheet = getSheet(); const rows = getAllRows(sheet); return corsResponse({ ok: true, data: rows }); } catch (err) { return corsResponse({ ok: false, error: err.message }); } } // ── POST — add / update / delete ───────────────────────────── function doPost(e) { try { const body = JSON.parse(e.postData.contents); const action = body.action; // "add" | "update" | "delete" const sheet = getSheet(); if (action === "add") { const record = body.record; record.id = record.id || Utilities.getUuid(); record.createdAt = new Date().toISOString(); sheet.appendRow(objToRow(record)); return corsResponse({ ok: true, record }); } if (action === "update") { const record = body.record; const rowIdx = findRowIndex(sheet, record.id); if (rowIdx === -1) return corsResponse({ ok: false, error: "Record not found" }); sheet.getRange(rowIdx, 1, 1, COLUMNS.length).setValues([objToRow(record)]); return corsResponse({ ok: true, record }); } if (action === "delete") { const rowIdx = findRowIndex(sheet, body.id); if (rowIdx === -1) return corsResponse({ ok: false, error: "Record not found" }); sheet.deleteRow(rowIdx); return corsResponse({ ok: true }); } return corsResponse({ ok: false, error: "Unknown action: " + action }); } catch (err) { return corsResponse({ ok: false, error: err.message }); } }
